/*
 * Pídele · Puente de WhatsApp (Cloudflare Worker)
 *
 * 1) Ruta alterna de la app (/api): reenvía las llamadas a Apps Script cuando una red no deja llegar directo a Google.
 * 2) Recibe los avisos de WhatsApp Business Platform (Meta), los pasa a Apps Script y contesta por WhatsApp.
 * Meta exige una respuesta directa a sus avisos y Apps Script siempre responde con una redirección;
 * este Worker hace de intermediario. Plan gratuito de Cloudflare: 100,000 avisos al día.
 *
 * Variables (Worker → Settings → Variables and Secrets):
 *   APPS_SCRIPT_URL  URL /exec de tu Apps Script (la misma de index.html)
 *   SECRETO          Texto largo inventado; el mismo que la propiedad WA_SECRETO en Apps Script
 *   VERIFY_TOKEN     Texto inventado que escribes en Meta al registrar el webhook
 *   WA_TOKEN         Token de acceso permanente de WhatsApp (usuario del sistema de Meta)
 *   WA_PHONE_ID      "Phone number ID" del número de Pídele en Meta
 *   APP_SECRET       "App secret" de tu app de Meta (para comprobar que el aviso viene de Meta)
 *   GRAPH_VERSION    Opcional, ej. v21.0
 */
const CORS = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', 'Access-Control-Max-Age': '86400' };

export default {
  async fetch(req, env, ctx) {
    const url = new URL(req.url);

    // 0) Ruta alterna de la app: si una red (ej. datos móviles) no deja llegar directo a Google, la app pasa por aquí.
    //    Solo reenvía a TU Apps Script (APPS_SCRIPT_URL) y devuelve su respuesta tal cual.
    if (url.pathname === '/api') {
      if (req.method === 'OPTIONS') return new Response(null, { status: 204, headers: CORS });
      if (req.method !== 'POST') return new Response('Pídele · ruta alterna activa', { status: 200, headers: CORS });
      try {
        const r = await fetch(env.APPS_SCRIPT_URL, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: await req.text(), redirect: 'follow' });
        return new Response(await r.text(), { status: 200, headers: Object.assign({ 'Content-Type': 'application/json; charset=utf-8', 'Cache-Control': 'no-store' }, CORS) });
      } catch (e) {
        return new Response(JSON.stringify({ ok: false, error: 'No se pudo conectar con el servidor. Intenta de nuevo.' }), { status: 200, headers: Object.assign({ 'Content-Type': 'application/json' }, CORS) });
      }
    }

    // 1) Meta comprueba el webhook una sola vez al registrarlo
    if (req.method === 'GET') {
      if (url.searchParams.get('hub.mode') === 'subscribe' && url.searchParams.get('hub.verify_token') === env.VERIFY_TOKEN) {
        return new Response(url.searchParams.get('hub.challenge') || '', { status: 200 });
      }
      return new Response('Pídele · puente de WhatsApp activo', { status: 200 });
    }
    if (req.method !== 'POST') return new Response('Método no permitido', { status: 405 });

    // 2) Aviso de mensaje: se comprueba la firma de Meta
    const cuerpo = await req.text();
    if (env.APP_SECRET && !(await firmaValida(cuerpo, req.headers.get('x-hub-signature-256') || '', env.APP_SECRET))) {
      return new Response('Firma inválida', { status: 401 });
    }
    let data;
    try { data = JSON.parse(cuerpo); } catch (e) { return new Response('JSON inválido', { status: 400 }); }

    // 3) Se responde a Meta al instante y los mensajes se procesan en segundo plano
    const mensajes = [];
    for (const entry of data.entry || []) {
      for (const ch of entry.changes || []) {
        for (const m of (ch.value && ch.value.messages) || []) {
          if (m.type === 'text' && m.text && m.text.body) mensajes.push(m);
        }
        // Avisos de estado de lo que Pídele envió: enviado, entregado, leído o falló (con motivo)
        for (const s of (ch.value && ch.value.statuses) || []) {
          const err = (s.errors || []).map(e => '(#' + e.code + ') ' + (e.title || '') + (e.error_data && e.error_data.details ? ' · ' + e.error_data.details : '')).join(' | ');
          console.log('Estado del mensaje a ' + s.recipient_id + ': ' + s.status + (err ? ' · ' + err : ''));
        }
      }
    }
    console.log('Aviso de Meta: ' + mensajes.length + ' mensaje(s) de texto');
    if (mensajes.length) ctx.waitUntil(Promise.all(mensajes.map(m => procesar(m, env))));
    return new Response('OK', { status: 200 });
  }
};

/** Pasa el mensaje a Apps Script y envía por WhatsApp la respuesta que devuelva. */
async function procesar(m, env) {
  try {
    const r = await fetch(env.APPS_SCRIPT_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'text/plain;charset=utf-8' },
      body: JSON.stringify({ accion: 'whatsappEntrante', datos: { secreto: env.SECRETO, de: m.from, texto: m.text.body, id: m.id } }),
      redirect: 'follow'
    });
    const txt = await r.text();
    let j;
    try { j = JSON.parse(txt); } catch (e) { console.log('Apps Script no devolvió JSON (¿URL /exec correcta y publicada?):', r.status, txt.slice(0, 200)); return; }
    if (!j.ok) { console.log('Apps Script respondió error:', j.error); return; }
    console.log('Apps Script OK, respondiendo a ' + m.from);
    if (j.data && j.data.respuesta) await responder(m.from, j.data.respuesta, env);
  } catch (e) {
    console.log('Error procesando mensaje:', e && e.message);
  }
}

/** Respuesta dentro de las 24 h después de que el usuario escribe: es gratis en WhatsApp Business Platform. */
async function responder(para, texto, env) {
  const v = env.GRAPH_VERSION || 'v21.0';
  // México: WhatsApp manda el número como 521 + 10 dígitos, pero para contestar Meta espera 52 + 10 dígitos
  if (/^521\d{10}$/.test(para)) para = '52' + para.slice(3);
  const r = await fetch(`https://graph.facebook.com/${v}/${env.WA_PHONE_ID}/messages`, {
    method: 'POST',
    headers: { Authorization: 'Bearer ' + env.WA_TOKEN, 'Content-Type': 'application/json' },
    body: JSON.stringify({ messaging_product: 'whatsapp', to: para, type: 'text', text: { body: texto } })
  });
  if (!r.ok) console.log('WhatsApp rechazó la respuesta', r.status, await r.text());
  else console.log('Respuesta enviada a ' + para);
}

/** Firma HMAC-SHA256 de Meta ("sha256=…") con el App secret. */
async function firmaValida(cuerpo, firma, secreto) {
  if (!firma.startsWith('sha256=')) return false;
  const llave = await crypto.subtle.importKey('raw', new TextEncoder().encode(secreto), { name: 'HMAC', hash: 'SHA-256' }, false, ['sign']);
  const sig = await crypto.subtle.sign('HMAC', llave, new TextEncoder().encode(cuerpo));
  const hex = [...new Uint8Array(sig)].map(b => b.toString(16).padStart(2, '0')).join('');
  const esperado = firma.slice(7);
  if (hex.length !== esperado.length) return false;
  let dif = 0;
  for (let i = 0; i < hex.length; i++) dif |= hex.charCodeAt(i) ^ esperado.charCodeAt(i);
  return dif === 0;
}
